async function getOrCreateCustomer(){ const customerName = customerNameInput.value.trim(); const customerPhone = customerPhoneInput.value.trim(); if(!customerName){ throw new Error( "Please enter customer name." ); } // ========================================== // CHECK CURRENT LOGIN SESSION // ========================================== const { data:sessionData, error:sessionError } = await supabaseClient.auth.getSession(); if(sessionError){ throw new Error( "Login session error: " + sessionError.message ); } if(!sessionData.session){ throw new Error( "Login session পাওয়া যায়নি। Logout করে আবার Login করুন." ); } console.log( "Authenticated user:", sessionData.session.user.email ); // ========================================== // FIND EXISTING CUSTOMER // ========================================== let customerQuery = supabaseClient .from("customers") .select("id,name,phone") .eq( "name", customerName ); if(customerPhone){ customerQuery = customerQuery.eq( "phone", customerPhone ); } const { data:existingCustomer, error:findError } = await customerQuery .limit(1); if(findError){ throw new Error( "Customer search failed: " + findError.message ); } // ========================================== // CUSTOMER ALREADY EXISTS // ========================================== if( existingCustomer && existingCustomer.length > 0 ){ console.log( "Existing Customer:", existingCustomer[0] ); return existingCustomer[0].id; } // ========================================== // CREATE NEW CUSTOMER // ========================================== const { data:newCustomer, error:createError } = await supabaseClient .from("customers") .insert({ name: customerName, phone: customerPhone }) .select("id") .single(); if(createError){ console.error( "CUSTOMER CREATE ERROR:", createError ); throw new Error( "Customer create failed: " + createError.message ); } if( !newCustomer || !newCustomer.id ){ throw new Error( "Customer ID পাওয়া যায়নি." ); } console.log( "New Customer ID:", newCustomer.id ); return newCustomer.id; }
